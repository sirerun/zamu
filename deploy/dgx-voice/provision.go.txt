package main
import("context";"encoding/json";"fmt";"os";"path/filepath";"strings";"github.com/zati-run/zati/internal/platform")
func main(){if err:=run();err!=nil{fmt.Fprintln(os.Stderr,err);os.Exit(1)}}
func run()error{
 if len(os.Args)!=3{return fmt.Errorf("usage: provision controller-root voice-root")}; root,voice:=os.Args[1],os.Args[2]
 p,err:=platform.Open(platform.Config{StateDir:filepath.Join(root,"state"),CredentialBackend:"headless",MasterKeyRef:"file:"+filepath.Join(root,"secrets/master.key")});if err!=nil{return err};defer p.Close()
 refs:=map[string]string{};for _,name:=range []string{"mistt","narrate"}{b,e:=os.ReadFile(filepath.Join(voice,"credentials",name,"token"));if e!=nil{return e};b=[]byte(strings.TrimSpace(string(b)));if len(b)<32{return fmt.Errorf("short credential")};ref,e:=p.Secrets().Put(context.Background(),"zati-voice-20261014-"+name,b);clear(b);if e!=nil{return e};refs[name]=ref}
 cfg:=map[string]string{"mistt_address":"127.0.0.1:18792","mistt_credential_ref":refs["mistt"],"narrate_origin":"http://127.0.0.1:18092","narrate_credential_ref":refs["narrate"]}
 b,e:=json.MarshalIndent(cfg,"","  ");if e!=nil{return e};f,e:=os.OpenFile(filepath.Join(voice,"controller.json"),os.O_CREATE|os.O_EXCL|os.O_WRONLY,0600);if e!=nil{return e};defer f.Close();_,e=f.Write(b);if e==nil{fmt.Println("Speech credentials stored; private controller config written")};return e
}
